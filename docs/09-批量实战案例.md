## 案例 3：新建 236 个品牌（L1 + L2 成对）

def create_brand(name, sort):
    # L1
    r1 = post("/admin/goodsCate/add", {
        "cate_name": name, "cate_pid": 0, "cate_type": 3,
        "cate_icon": "", "shop_type": 1, "is_display": 1,
        "period": 72, "sort": sort, "state": 1,
    })
    if r1.get("code") != 0:
        return False
    # 🔴 add 不返回 id，必须重新 lists 反查
    time.sleep(0.3)
    brands = post("/admin/goodsCate/lists", {"cate_type": 3, "page": 1, "limit": 600})["data"]
    l1_id = next((b["cate_id"] for b in brands
                  if b.get("cate_pid") == 0 and b["cate_name"] == name), None)
    if not l1_id:
        return False
    # 同名 L2
    r2 = post("/admin/goodsCate/add", {
        "cate_name": name, "cate_pid": l1_id, "cate_type": 3,
        "cate_icon": "", "shop_type": 1, "is_display": 1,
        "period": 72, "sort": sort, "state": 1,
    })
    return r2.get("code") == 0

# 按权重档位建
for name, lvl in BRAND_LIB_ITEMS:

## 案例 4：1534 个商品迁移到新二级

```python
def batch_change_cate(changes, sleep=0.12):
    ok = fail = 0
    for i, c in enumerate(changes, 1):
        try:
            detail = post("/admin/good/detail", {"good_id": c["good_id"]}).get("data") or {}
            body = build_body(detail, [[1, c["new_c1_id"], c["new_c2_id"]]])   # 见 03 章
            r = post(f"/admin/good/edit?good_id={c['good_id']}", body)
            if r.get("code") == 0:
                ok += 1
            else:
                fail += 1
                print(f"✗ {c['good_id']}: {r.get('msg')}")
            time.sleep(sleep)     # 🔴 必须节流
        except Exception as e:
            fail += 1
            print(f"ERR {c['good_id']}: {e}")
        if i % 100 == 0:
            print(f"进度 {i}/{len(changes)} ok={ok} fail={fail}", flush=True)
    return ok, fail
```

**性能**：1098 条 → 约 8 分钟；1534 条全量 → 约 12 分钟。

## 案例 7：商品归类纠错（关键词 + 黑名单）

```python
# 严格关键词（避免误判）+ 黑名单否定
STRICT_KEYWORD = {
    '铅笔': ['铅笔', '素描铅'],
    '纸品': ['打印纸', '复印纸', '相纸', '相片纸', '不干胶标签', '热敏标签', '条码标签'],
    '儿童牙膏': ['儿童牙膏', '宝宝牙膏', '幼儿牙膏', '可吞咽牙膏'],
    # ...
}

# 黑名单：命中这些词则不触发对应二级
NEG = {
    '铅笔': ['削笔', '笔刀'],           # 卷笔刀含"削笔"，不能归铅笔
    '作业本': ['中性笔', '礼盒', '双肩包'],  # 笔记本礼盒不能归作业本
    '纸品': ['热敏纸', '收银纸'],        # 牛皮纸手工纸不算纸品
}

def recat(name):
    for sub, kws in sorted(STRICT_KEYWORD.items(), key=lambda x: -max(len(k) for k in x[1])):
        if any(n in name for n in NEG.get(sub, [])):
            continue
        for kw in kws:
            if kw in name:
                return sub
    return None
```

**效果**：干跑 125 个候选 → 黑名单精筛到 30 个 → **30/30 成功**。

### 关键词设计四原则

1. **只匹配强信号**：`儿童牛奶`/`成长奶`，不用单字"奶"
2. **黑名单否定**：`削笔` 不触发 `铅笔`
3. **长关键词优先**：`儿童氨基酸洗发水`(10字) > `洗发水`(3字)
4. **避免单字触发**："牛奶"会被"牛奶杯"误判
